/* ============================================================ ROMPECABEZAS EL REY JavaScript Vanilla ============================================================ */ (() => { "use strict"; /* ------------------------------------------------------------ CONFIGURACIÓN ------------------------------------------------------------ */ const IMAGES = [ "imagen1.jpeg", "imagen2.jpeg", "imagen3.jpeg" ]; /* * IMPORTANTE: * * Firebase Realtime Database NO utiliza normalmente * "https://firebaseio.com" como URL de base de datos. * * Debés reemplazar este valor por la URL de TU Realtime * Database, por ejemplo: * * https://tu-proyecto-default-rtdb.firebaseio.com * * Se mantiene separado para que no haya que modificar * ninguna otra parte del juego. */ const FIREBASE_URL = "https://firebaseio.com"; let pieces = []; let solved = false; let winningCode = "0000"; let activeTouch = null; let activeMouse = null; /* ------------------------------------------------------------ UTILIDADES ------------------------------------------------------------ */ function random(min, max) { return Math.random() * (max - min) + min; } function randomInt(min, max) { return Math.floor(random(min, max + 1)); } function clamp(value, min, max) { return Math.max(min, Math.min(max, value)); } function formatCode(number) { return String(number).padStart(4, "0"); } function setStatus(message) { statusElement.textContent = message || ""; } function chooseImage(forceDifferent = false) { if (!forceDifferent || IMAGES.length <= 1 || !currentImage) { currentImage = IMAGES[randomInt(0, IMAGES.length - 1)]; El fondo completo es exactamente: 330px x 330px Cada pieza desplaza el fondo: columna * -110px fila * -110px ------------------------------------------------------------ */ function applyPieceBackground(piece, index) { const row = Math.floor(index / GRID_SIZE); const column = index % GRID_SIZE; const x = column * -PIECE_SIZE; const y = row * -PIECE_SIZE; piece.style.backgroundImage = `url("${currentImage}")`; piece.style.backgroundSize = `${PUZZLE_SIZE}px ${PUZZLE_SIZE}px`; piece.style.backgroundPosition = `${x}px ${y}px`; piece.style.backgroundRepeat = "no-repeat"; } /* ------------------------------------------------------------ CREAR PIEZAS ------------------------------------------------------------ */ function createPieces() { pieces = []; for (let index = 0; index < 9; index++) { const piece = document.createElement("div"); piece.className = `piece p${index}`; piece.dataset.index = String(index); applyPieceBackground(piece, index); /* * Cada pieza empieza abajo, desordenada. */ arena.appendChild(piece); pieces.push({ element: piece, index, placed: false, locked: false, location: "arena", x: 0, y: 0, rotation: 0 }); setupTouchEvents(piece, index); setupMouseEvents(piece, index); } } /* ------------------------------------------------------------ DISPERSIÓN INICIAL ------------------------------------------------------------ */ function scatterPieces() { const arenaWidth = arena.clientWidth; pieces.forEach(item => { const piece = item.element; const maxX = Math.max(0, arenaWidth - PIECE_SIZE); const maxY = Math.max(0, ARENA_HEIGHT - PIECE_SIZE); const x = random(0, maxX); const y = random(5, maxY); const rotation = random(-12, 12); item.x = x; item.y = y; item.rotation = rotation; item.location = "arena"; item.placed = false; item.locked = false; piece.classList.remove("dragging", "locked"); piece.style.left = `${x}px`; const column = index % GRID_SIZE; return { x: column * PIECE_SIZE, y: row * PIECE_SIZE }; } /* ------------------------------------------------------------ DETECTAR SI LA PIEZA FUE SOLTADA SOBRE EL TABLERO ------------------------------------------------------------ */ function getDropCell(clientX, clientY) { const rect = board.getBoundingClientRect(); const arenaRect = arena.getBoundingClientRect(); /* * Convertimos la coordenada de pantalla al sistema * absoluto de la arena. */ let x = clientX - arenaRect.left - PIECE_SIZE / 2; let y = clientY - arenaRect.top - PIECE_SIZE / 2; x = clamp( x, 0, Math.max(0, arena.clientWidth - PIECE_SIZE) ); y = clamp( y, 0, Math.max(0, ARENA_HEIGHT - PIECE_SIZE) ); item.x = x; item.y = y; item.rotation = random(-12, 12); item.location = "arena"; item.placed = false; item.locked = false; item.element.classList.remove("locked"); item.element.style.left = `${x}px`; item.element.style.top = `${y}px`; item.element.style.transform = `rotate(${item.rotation}deg)`; arena.appendChild(item.element); } /* ------------------------------------------------------------ SOLTAR EN TABLERO REGLA IMPORTANTE: Si es incorrecta, NO vuelve automáticamente abajo. Se queda fija en esa casilla. ------------------------------------------------------------ */ function dropOnBoard(item, clientX, clientY) { const cell = getDropCell(clientX, clientY); if (!cell) { return false; } item.x = cell.x; item.y = cell.y; item.location = "board"; item.element.style.left = `${cell.x}px`; item.element.style.top = `${cell.y}px`; item.element.style.transform = "rotate(0deg)"; /* * La pieza queda en el tablero independientemente de que * sea correcta o incorrecta. */ board.appendChild(item.element); if (cell.index === item.index) { item.placed = true; item.locked = true; item.element.classList.add("locked"); } else { /* * ERROR INTENCIONAL: * La pieza permanece en la casilla incorrecta. * * NO se devuelve a la arena. * Sigue siendo arrastrable porque locked = false. */ item.placed = false; item.locked = false; item.element.classList.remove("locked"); } checkVictory(); return true; } /* ------------------------------------------------------------ PROCESAR SOLTADO ------------------------------------------------------------ */ function processDrop(item, clientX, clientY) { if (!item || item.locked || solved) { return; } const cell = getDropCell(clientX, clientY); if (cell) { dropOnBoard(item, clientX, clientY); } else { returnToArena(item, clientX, clientY); } } /* ------------------------------------------------------------ OBTENER PIEZA ------------------------------------------------------------ */ function getPiece(index) { return pieces.find(item => item.index === index); } /* ------------------------------------------------------------ TOUCHSTART Evento exclusivamente táctil. ------------------------------------------------------------ */ function handleTouchStart(e, index) { if (solved) return; const item = getPiece(index); if (!item || item.locked) { return; } if (!e.touches || !e.touches.length) { return; } const touch = e.touches[0]; const rect = item.element.getBoundingClientRect(); activeTouch = { item, identifier: touch.identifier, offsetX: touch.clientX - rect.left, offsetY: touch.clientY - rect.top }; item.element.classList.add("dragging"); /* * Se quita la rotación durante el arrastre. */ item.element.style.transform = "rotate(0deg)"; /* * Elevamos visualmente la pieza. */ item.element.style.zIndex = "9999"; } /* ------------------------------------------------------------ TOUCHMOVE Evento exclusivamente táctil. ------------------------------------------------------------ */ function handleTouchMove(e) { if (!activeTouch) { return; } let touch = null; for (const currentTouch of e.touches) { if ( currentTouch.identifier === activeTouch.identifier ) { touch = currentTouch; break; } } if (!touch) { return; } /* * Evita que el navegador haga scroll mientras arrastramos. */ e.preventDefault(); const item = activeTouch.item; let parentRect; if (item.location === "board") { parentRect = board.getBoundingClientRect(); } else { parentRect = arena.getBoundingClientRect(); } let x = touch.clientX - parentRect.left - activeTouch.offsetX; let y = touch.clientY - parentRect.top - activeTouch.offsetY; /* * Permitimos mover la pieza libremente, incluso fuera * del contenedor. */ item.element.style.left = `${x}px`; item.element.style.top = `${y}px`; item.x = x; item.y = y; } /* ------------------------------------------------------------ TOUCHEND Evento exclusivamente táctil. ------------------------------------------------------------ */ function handleTouchEnd(e) { if (!activeTouch) { return; } let touch = null; if ( e.changedTouches && e.changedTouches.length ) { for (const currentTouch of e.changedTouches) { if ( currentTouch.identifier === activeTouch.identifier ) { touch = currentTouch; break; } } } const item = activeTouch.item; if (touch) { processDrop( item, touch.clientX, touch.clientY ); } item.element.classList.remove("dragging"); if (!item.locked) { item.element.style.zIndex = "20"; } else { item.element.style.zIndex = "50"; } activeTouch = null; } /* ------------------------------------------------------------ REGISTRAR TOUCH EVENTS ------------------------------------------------------------ */ function setupTouchEvents(element, index) { element.addEventListener( "touchstart", e => handleTouchStart(e, index), { passive: false } ); element.addEventListener( "touchmove", handleTouchMove, { passive: false } ); element.addEventListener( "touchend", handleTouchEnd, { passive: false } ); element.addEventListener( "touchcancel", handleTouchEnd, { passive: false } ); } /* ------------------------------------------------------------ MOUSE DOWN Separado completamente de TouchEvent. ------------------------------------------------------------ */ function handleMouseDown(e, index) { if (solved) return; const item = getPiece(index); if (!item || item.locked) { return; } const rect = item.element.getBoundingClientRect(); activeMouse = { item, offsetX: e.clientX - rect.left, offsetY: e.clientY - rect.top }; item.element.classList.add("dragging"); item.element.style.transform = "rotate(0deg)"; item.element.style.zIndex = "9999"; e.preventDefault(); } /* ------------------------------------------------------------ MOUSE MOVE ------------------------------------------------------------ */ function handleMouseMove(e) { if (!activeMouse) { return; } const item = activeMouse.item; const parentRect = item.location === "board" ? board.getBoundingClientRect() : arena.getBoundingClientRect(); const x = e.clientX - parentRect.left - activeMouse.offsetX; const y = e.clientY - parentRect.top - activeMouse.offsetY; item.element.style.left = `${x}px`; item.element.style.top = `${y}px`; item.x = x; item.y = y; } /* ------------------------------------------------------------ MOUSE UP ------------------------------------------------------------ */ function handleMouseUp(e) { if (!activeMouse) { return; } const item = activeMouse.item; processDrop( item, e.clientX, e.clientY ); item.element.classList.remove("dragging"); if (!item.locked) { item.element.style.zIndex = "20"; } else { item.element.style.zIndex = "50"; } activeMouse = null; } /* ------------------------------------------------------------ REGISTRAR MOUSE EVENTS ------------------------------------------------------------ */ function setupMouseEvents(element, index) { element.addEventListener( "mousedown", e => handleMouseDown(e, index) ); } document.addEventListener( "mousemove", handleMouseMove ); document.addEventListener( "mouseup", handleMouseUp ); /* ------------------------------------------------------------ VALIDACIÓN DEL ROMPECABEZAS ------------------------------------------------------------ */ function validatePuzzle() { if (pieces.length !== 9) { return false; } for (let index = 0; index < 9; index++) { const item = getPiece(index); if (!item) { return false; } if (!item.placed) { return false; } if (!item.locked) { return false; } /* * Validación adicional de posición: */ const expected = getBoardPosition(index); const actualLeft = parseFloat(item.element.style.left) || 0; const actualTop = parseFloat(item.element.style.top) || 0; if ( Math.abs(actualLeft - expected.x) > 1 || Math.abs(actualTop - expected.y) > 1 ) { return false; } } return true; } function checkVictory() { if (solved) { return; } if (!validatePuzzle()) { return; } solved = true; /* * Pequeño retraso para permitir que el usuario vea * la última pieza colocada. */ setTimeout(() => { registerWinner(); }, 250); } /* ------------------------------------------------------------ FIREBASE POST: registra una victoria. GET: obtiene todas las claves actuales. nocache: evita respuestas almacenadas en caché. ------------------------------------------------------------ */ async function registerWinner() { setStatus("Registrando victoria..."); try { const firebaseBase = FIREBASE_URL.replace(/\/+$/, ""); const postUrl = `${firebaseBase}${FIREBASE_NODE}`; /* * POST crea una nueva clave automática. */ const postResponse = await fetch( postUrl, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ timestamp: Date.now(), game: "rompecabezas-el-rey" }), cache: "no-store" } ); if (!postResponse.ok) { throw new Error( `Firebase POST ${postResponse.status}` ); } /* * GET limpio con parámetro único de caché. */ const getUrl = `${firebaseBase}${FIREBASE_NODE}?nocache=${Date.now()}`; const getResponse = await fetch( getUrl, { method: "GET", cache: "no-store", headers: { "Cache-Control": "no-cache" } } ); if (!getResponse.ok) { throw new Error( `Firebase GET ${getResponse.status}` ); } const data = await getResponse.json(); let total = 0; if ( data && typeof data === "object" && !Array.isArray(data) ) { total = Object.keys(data).length; } winningCode = formatCode(total); showVictory(winningCode); } catch (error) { console.error( "Error registrando victoria:", error ); /* * No inventamos un código si Firebase falló. * Mostramos un mensaje para que el problema sea visible. */ setStatus( "No se pudo conectar con el contador. Revisá la URL de Firebase." ); solved = false; } } /* ------------------------------------------------------------ MODAL DE VICTORIA ------------------------------------------------------------ */ function showVictory(code) { counterCode.textContent = code; victoryOverlay.classList.add("show"); setStatus(""); } /* ------------------------------------------------------------ BOTÓN COBRAR ------------------------------------------------------------ */ function claimPrize() { const phone = "5491124050658"; const message = `felicitaciones por finalizar el rompecabezas. Mi código es: ${winningCode}`; const whatsappUrl = `https://wa.me/${phone}?text=${encodeURIComponent(message)}`; window.location.href = whatsappUrl; } claimButton.addEventListener( "click", claimPrize ); /* ------------------------------------------------------------ REINICIAR ------------------------------------------------------------ */ function resetGame() { solved = false; winningCode = "0000"; victoryOverlay.classList.remove("show"); /* * Selecciona una imagen diferente a la anterior * siempre que haya más de una disponible. */ chooseImage(true); /* * Eliminamos las piezas antiguas. */ pieces.forEach(item => { item.element.remove(); }); pieces = []; /* * Creamos nuevamente las 9 piezas con los fondos * correspondientes a sus índices. */ createPieces(); /* * Esperamos un frame para asegurar que arena.clientWidth * esté calculado correctamente. */ requestAnimationFrame(() => { scatterPieces(); }); setStatus(""); } resetButton.addEventListener( "click", resetGame ); /* ------------------------------------------------------------ INICIALIZACIÓN ------------------------------------------------------------ */ function init() { chooseImage(false); createPieces(); requestAnimationFrame(() => { scatterPieces(); }); } init(); })();
